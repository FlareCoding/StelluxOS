# Arduino-ESP32 3.3.12's recipes as Stellux runs them: mcufw's hook verbs in place of its bash hooks, esptool run by
# python3, and ar, since gcc-ar needs the LTO plugin loader the toolchain package leaves out
compiler.ar.cmd={compiler.prefix}ar
# One copy-first does the work of prebuild.1 to .3, as another does of .5 and .6

recipe.hooks.prebuild.2.pattern=
recipe.hooks.prebuild.3.pattern=

recipe.hooks.prebuild.5.pattern=mcufw hook copy-first "{build.path}/build_opt.h" "{build.source.path}/build_opt.h" --else mcufw hook write "{build.path}/build_opt.h"
recipe.hooks.prebuild.6.pattern=
recipe.hooks.prebuild.7.pattern=mcufw hook write "{file_opts.path}"

recipe.hooks.core.prebuild.1.pattern=mcufw hook write "{file_opts.path}" -DARDUINO_CORE_BUILD
recipe.hooks.core.postbuild.1.pattern=mcufw hook write "{file_opts.path}"
